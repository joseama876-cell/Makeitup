 * "Make it yours. Make it up."
 */
import React, { useState, useEffect, useMemo } from 'react';
import {
  ShoppingBag,
  Heart,
  Search,
  Menu,
  X,
  Star,
  ArrowRight,
  ChevronLeft,
  ChevronRight,
  ShieldCheck,
  Truck,
  RotateCcw,
  Sparkles,
  CheckCircle2,
  Package,
  Clock,
  MessageCircle,
  SlidersHorizontal,
  Mail,
  ExternalLink,
  Info,
  MapPin,
  Check
} from 'lucide-react';
import { PRODUCTS, CATEGORIES_NAV, INSTAGRAM_POSTS, REVIEWS } from './data/products';
import { Product, CartItem, Category, OrderCustomerDetails, OrderRecord } from './types';

const ORDER_TARGET_EMAIL = "josephamalraj.co@gmail.com";
const WHATSAPP_PHONE = "919876543210";

// Pre-seeded sample order so Track Order works right away out of the box
const SAMPLE_ORDER: OrderRecord = {
  orderId: "MIU-849201",
  orderDate: "05 Oct 2026, 06:45 PM",
  status: "Shipped",
  estimatedDelivery: "08 Oct 2026",
  customer: {
    name: "Vikram Singhania",
    phone: "9876543210",
    email: "vikram.singh@gmail.com",
    address: "B-402, Royal Palms, Bandra West",
    city: "Mumbai",
    pincode: "400050",
    notes: "Leave with security if not answering",
    paymentMethod: "UPI"
  },
  items: [
    {
      product: PRODUCTS[2], // Festive Gold Kurta
      selectedSize: "L",
      quantity: 1
    },
    {
      product: PRODUCTS[10], // Leather Belt
      selectedSize: "Free Size (30-40)",
      quantity: 1
    }
  ],
  subtotal: 3098,
  discount: 310,
  shipping: 0,
  total: 2788,
  courierName: "Blue Dart Express",
  trackingNumber: "BD-8893049102"
};

export default function App() {
  // Navigation & Modal states
  const [isMobileMenuOpen, setIsMobileMenuOpen] = useState(false);
  const [isCartOpen, setIsCartOpen] = useState(false);
  const [isWishlistOpen, setIsWishlistOpen] = useState(false);
  const [isSearchOpen, setIsSearchOpen] = useState(false);
  const [isCheckoutOpen, setIsCheckoutOpen] = useState(false);
  const [isTrackOrderOpen, setIsTrackOrderOpen] = useState(false);
  const [selectedProduct, setSelectedProduct] = useState<Product | null>(null);

  // Search & Catalog Filter states
  const [searchQuery, setSearchQuery] = useState('');
  const [selectedCategory, setSelectedCategory] = useState<Category>('All');
  const [sortBy, setSortBy] = useState<'featured' | 'price-low' | 'price-high' | 'rating'>('featured');

  // Track Order query state
  const [trackingInputId, setTrackingInputId] = useState('');
  const [trackedOrder, setTrackedOrder] = useState<OrderRecord | null>(null);
  const [trackingSearchError, setTrackingSearchError] = useState('');

  // Cart & Wishlist persistence (localStorage)
  const [cart, setCart] = useState<CartItem[]>(() => {
    try {
      const saved = localStorage.getItem('makeitup_cart');
      return saved ? JSON.parse(saved) : [];
    } catch {
      return [];
    }
  });

  const [wishlist, setWishlist] = useState<number[]>(() => {
    try {
      const saved = localStorage.getItem('makeitup_wishlist');
      return saved ? JSON.parse(saved) : [1, 3];
    } catch {
      return [1, 3];
    }
  });

  // Orders stored in localStorage
  const [savedOrders, setSavedOrders] = useState<OrderRecord[]>(() => {
    try {
      const saved = localStorage.getItem('makeitup_orders');
      if (saved) {
        const parsed = JSON.parse(saved);
        if (Array.isArray(parsed) && parsed.length > 0) return parsed;
      }
      return [SAMPLE_ORDER];
    } catch {
      return [SAMPLE_ORDER];
    }
  });

  useEffect(() => {
    localStorage.setItem('makeitup_cart', JSON.stringify(cart));
  }, [cart]);

  useEffect(() => {
    localStorage.setItem('makeitup_wishlist', JSON.stringify(wishlist));
  }, [wishlist]);

  useEffect(() => {
    localStorage.setItem('makeitup_orders', JSON.stringify(savedOrders));
  }, [savedOrders]);

  // Hero Carousel Slides
  const [heroSlide, setHeroSlide] = useState(0);
  const heroSlides = [
    {
      subtitle: "The Royal Festive Collection 2026",
      title: "Festive Essentials – Celebrate in Style",
      accent: "Handcrafted Zari & Raw Silk",
      bg: "https://images.unsplash.com/photo-1617137984095-74e4e5e3613f?w=1600&q=80",
      cta: "EXPLORE FESTIVE EDIT",
      category: "Ethnic" as Category
    },
    {
      subtitle: "New Drop • Limited Run",
      title: "Artisanal Graphic & Vintage Shirts",
      accent: "Cuban Collars & Heritage Weaves",
      bg: "https://images.unsplash.com/photo-1596755094514-f87e34085b2c?w=1600&q=80",
      cta: "SHOP SHIRTS",
      category: "Shirt" as Category
    },
    {
      subtitle: "Editorial Sartorial Tailoring",
      title: "Understated Elegance for Everyday",
      accent: "French Linen & Pleated Trousers",
      bg: "https://images.unsplash.com/photo-1507679799987-c73779587ccf?w=1600&q=80",
      cta: "VIEW ALL MENSWEAR",
      category: "All" as Category
    }
  ];

  useEffect(() => {
    const timer = setInterval(() => {
      setHeroSlide((prev) => (prev + 1) % heroSlides.length);
    }, 5500);
    return () => clearInterval(timer);
  }, [heroSlides.length]);

  // Coupon state
  const [couponCode, setCouponCode] = useState('');
  const [appliedCoupon, setAppliedCoupon] = useState<{ code: string; discountPercent: number } | null>(null);
  const [couponError, setCouponError] = useState('');

  // Checkout Form
  const [checkoutForm, setCheckoutForm] = useState<OrderCustomerDetails>({
    name: '',
    phone: '',
    email: '',
    address: '',
    city: '',
    pincode: '',
    notes: '',
    paymentMethod: 'COD'
  });
  const [formErrors, setFormErrors] = useState<Record<string, string>>({});
  const [isSubmittingOrder, setIsSubmittingOrder] = useState(false);
  const [placedOrder, setPlacedOrder] = useState<OrderRecord | null>(null);

  // Newsletter
  const [newsletterEmail, setNewsletterEmail] = useState('');
  const [newsletterSubscribed, setNewsletterSubscribed] = useState(false);

  // Cart Calculations
  const cartSubtotal = useMemo(() => {
    return cart.reduce((sum, item) => sum + item.product.price * item.quantity, 0);
  }, [cart]);

  const discountAmount = useMemo(() => {
    if (!appliedCoupon) return 0;
    return Math.round((cartSubtotal * appliedCoupon.discountPercent) / 100);
  }, [cartSubtotal, appliedCoupon]);

  const shippingFee = cartSubtotal >= 1999 || cartSubtotal === 0 ? 0 : 99;
  const cartTotal = Math.max(0, cartSubtotal - discountAmount + shippingFee);
  const totalCartItems = cart.reduce((count, item) => count + item.quantity, 0);

  // Wishlist actions
  const toggleWishlist = (productId: number, e?: React.MouseEvent) => {
    if (e) e.stopPropagation();
    setWishlist((prev) =>
      prev.includes(productId) ? prev.filter((id) => id !== productId) : [...prev, productId]
    );
  };

  // Cart actions
  const addToCart = (product: Product, size?: string) => {
    const chosenSize = size || product.sizes[0];
    setCart((prev) => {
      const existing = prev.find((item) => item.product.id === product.id && item.selectedSize === chosenSize);
      if (existing) {
        return prev.map((item) =>
          item.product.id === product.id && item.selectedSize === chosenSize
            ? { ...item, quantity: item.quantity + 1 }
            : item
        );
      }
      return [...prev, { product, selectedSize: chosenSize, quantity: 1 }];
    });
    setIsCartOpen(true);
  };

  const updateCartQuantity = (productId: number, size: string, delta: number) => {
    setCart((prev) =>
      prev
        .map((item) => {
          if (item.product.id === productId && item.selectedSize === size) {
            const newQty = item.quantity + delta;
            return newQty > 0 ? { ...item, quantity: newQty } : null;
          }
          return item;
        })
        .filter(Boolean) as CartItem[]
    );
  };

  const removeCartItem = (productId: number, size: string) => {
    setCart((prev) => prev.filter((item) => !(item.product.id === productId && item.selectedSize === size)));
  };

  const applyCoupon = () => {
    setCouponError('');
    const code = couponCode.trim().toUpperCase();
    if (code === 'MAKEITUP10' || code === 'WELCOME10') {
      setAppliedCoupon({ code, discountPercent: 10 });
    } else if (code === 'FESTIVE15') {
      setAppliedCoupon({ code, discountPercent: 15 });
    } else {
      setCouponError('Invalid coupon. Try MAKEITUP10 for 10% off.');
    }
  };

  // Filter & Search
  const filteredProducts = useMemo(() => {
    let list = [...PRODUCTS];

    if (selectedCategory !== 'All') {
      list = list.filter((p) => p.category === selectedCategory);
    }

    if (searchQuery.trim()) {
      const q = searchQuery.toLowerCase();
      list = list.filter(
        (p) =>
          p.name.toLowerCase().includes(q) ||
          p.category.toLowerCase().includes(q) ||
          p.description.toLowerCase().includes(q)
      );
    }

    if (sortBy === 'price-low') {
      list.sort((a, b) => a.price - b.price);
    } else if (sortBy === 'price-high') {
      list.sort((a, b) => b.price - a.price);
    } else if (sortBy === 'rating') {
      list.sort((a, b) => b.rating - a.rating);
    }

    return list;
  }, [selectedCategory, searchQuery, sortBy]);

  // Validation
  const validateCheckout = () => {
    const errors: Record<string, string> = {};
    if (!checkoutForm.name.trim()) errors.name = 'Full name is required';
    if (!checkoutForm.phone.trim() || !/^\d{10}$/.test(checkoutForm.phone.replace(/\D/g, ''))) {
      errors.phone = 'Valid 10-digit mobile number required';
    }
    if (!checkoutForm.email.trim() || !/\S+@\S+\.\S+/.test(checkoutForm.email)) {
      errors.email = 'Valid email address required';
    }
    if (!checkoutForm.address.trim()) errors.address = 'Delivery address is required';
    if (!checkoutForm.city.trim()) errors.city = 'City is required';
    if (!checkoutForm.pincode.trim() || !/^\d{6}$/.test(checkoutForm.pincode.replace(/\D/g, ''))) {
      errors.pincode = 'Valid 6-digit postal pincode required';
    }
    setFormErrors(errors);
    return Object.keys(errors).length === 0;
  };

  // Place Order & Dispatch Email
  const handlePlaceOrder = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!validateCheckout()) return;
    if (cart.length === 0) return;

    setIsSubmittingOrder(true);

    const generatedOrderId = `MIU-${Math.floor(100000 + Math.random() * 900000)}`;
    const now = new Date();
    const formattedDate = now.toLocaleDateString('en-IN', {
      day: '2-digit',
      month: 'short',
      year: 'numeric'
    }) + ', ' + now.toLocaleTimeString('en-IN', {
      hour: '2-digit',
      minute: '2-digit'
    });

    const estDate = new Date();
    estDate.setDate(estDate.getDate() + 4);
    const formattedEst = estDate.toLocaleDateString('en-IN', {
      day: '2-digit',
      month: 'short',
      year: 'numeric'
    });

    const newOrder: OrderRecord = {
      orderId: generatedOrderId,
      orderDate: formattedDate,
      status: 'Confirmed',
      estimatedDelivery: formattedEst,
      customer: checkoutForm,
      items: [...cart],
      subtotal: cartSubtotal,
      discount: discountAmount,
      shipping: shippingFee,
      total: cartTotal,
      courierName: "Blue Dart Express",
      trackingNumber: `BD-${Math.floor(1000000000 + Math.random() * 9000000000)}`
    };

    setSavedOrders((prev) => [newOrder, ...prev]);

    const itemsFormattedText = newOrder.items
      .map((i, idx) => `${idx + 1}. ${i.product.name} (Size: ${i.selectedSize}) x ${i.quantity} = ₹${i.product.price * i.quantity}`)
      .join('\n');

    const emailPayload = {
      _subject: `New makeitup Order #${newOrder.orderId} from ${newOrder.customer.name}`,
      order_id: newOrder.orderId,
      order_date: newOrder.orderDate,
      customer_name: newOrder.customer.name,
      customer_email: newOrder.customer.email,
      customer_phone: newOrder.customer.phone,
      delivery_address: `${newOrder.customer.address}, ${newOrder.customer.city} - ${newOrder.customer.pincode}`,
      customer_notes: newOrder.customer.notes || 'None',
      payment_method: newOrder.customer.paymentMethod,
      order_items: itemsFormattedText,
      subtotal: `₹${newOrder.subtotal}`,
      discount: `₹${newOrder.discount}`,
      shipping: `₹${newOrder.shipping}`,
      grand_total: `₹${newOrder.total}`
    };

    try {
      await fetch(`https://formsubmit.co/ajax/${ORDER_TARGET_EMAIL}`, {
        method: "POST",
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json'
        },
        body: JSON.stringify(emailPayload)
      });
    } catch (err) {
      console.warn('Email dispatch handled safely:', err);
    } finally {
      setIsSubmittingOrder(false);
      setPlacedOrder(newOrder);
      setCart([]);
      setIsCheckoutOpen(false);
    }
  };

  // Track Order query
  const handleSearchOrder = (idToSearch?: string) => {
    const query = (idToSearch || trackingInputId).trim().toUpperCase();
    if (!query) {
      setTrackingSearchError('Please enter an Order ID');
      setTrackedOrder(null);
      return;
    }

    setTrackingSearchError('');
    const found = savedOrders.find((o) => o.orderId.toUpperCase() === query);

    if (found) {
      setTrackedOrder(found);
      setTrackingInputId(found.orderId);
    } else {
      setTrackedOrder(null);
      setTrackingSearchError(`No order found with ID "${query}". Please check your order confirmation.`);
    }
  };

  return (
    <div className="min-h-screen bg-[#FAF7F0] text-[#111111] flex flex-col font-sans">
      
      {/* Top Banner Notice */}
      <div className="bg-[#111111] text-[#E0A800] text-xs font-medium py-2 px-4 text-center tracking-wider uppercase flex items-center justify-center gap-2">
        <Sparkles size={14} className="text-[#E0A800]" />
        <span>Festive Season Sale: Use code <strong className="underline text-white">MAKEITUP10</strong> for 10% OFF • Free Shipping on orders over ₹1,999</span>
      </div>

      {/* STICKY HEADER */}
      <header className="sticky top-0 z-40 bg-[#FAF7F0]/95 backdrop-blur-md border-b border-[#E8E2D5] transition-all">
        <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 h-18 flex items-center justify-between">
          
          {/* Left: Mobile Menu & Desktop Nav */}
          <div className="flex items-center gap-4">
            <button
              onClick={() => setIsMobileMenuOpen(!isMobileMenuOpen)}
              className="p-2 -ml-2 rounded-lg text-[#111111] hover:bg-black/5 md:hidden"
              aria-label="Open menu"
            >
              <Menu size={22} />
            </button>

            <nav className="hidden md:flex items-center gap-6 text-sm font-medium tracking-wide">
              <button
                onClick={() => { setSelectedCategory('All'); window.scrollTo({ top: 600, behavior: 'smooth' }); }}
                className="hover:text-[#E0A800] transition-colors uppercase text-xs tracking-widest font-semibold"
              >
                Collection
              </button>
              <button
                onClick={() => { setSelectedCategory('Shirt'); window.scrollTo({ top: 600, behavior: 'smooth' }); }}
                className="hover:text-[#E0A800] transition-colors uppercase text-xs tracking-widest"
              >
                Shirts
              </button>
              <button
                onClick={() => { setSelectedCategory('Ethnic'); window.scrollTo({ top: 600, behavior: 'smooth' }); }}
                className="hover:text-[#E0A800] transition-colors uppercase text-xs tracking-widest text-[#E0A800] font-bold"
              >
                Festive Kurta ✨
              </button>
              <button
                onClick={() => { setSelectedCategory('Trousers'); window.scrollTo({ top: 600, behavior: 'smooth' }); }}
                className="hover:text-[#E0A800] transition-colors uppercase text-xs tracking-widest"
              >
                Trousers
              </button>
              <button
                onClick={() => { setSelectedCategory('Accessories'); window.scrollTo({ top: 600, behavior: 'smooth' }); }}
                className="hover:text-[#E0A800] transition-colors uppercase text-xs tracking-widest"
              >
                Accessories
              </button>
              <button
                onClick={() => {
                  setIsTrackOrderOpen(true);
                  if (savedOrders.length > 0 && !trackedOrder) {
                    setTrackedOrder(savedOrders[0]);
                    setTrackingInputId(savedOrders[0].orderId);
                  }
                }}
                className="hover:text-[#E0A800] transition-colors uppercase text-xs tracking-widest font-bold flex items-center gap-1.5 text-black bg-[#FAF7F0] px-2.5 py-1 rounded-full border border-black/15"
              >
                <Truck size={14} className="text-[#E0A800]" /> Track Order
              </button>
            </nav>
          </div>

          {/* Center: Brand Wordmark */}
          <div className="text-center cursor-pointer" onClick={() => { setSelectedCategory('All'); window.scrollTo({ top: 0, behavior: 'smooth' }); }}>
            <span className="font-serif text-2xl sm:text-3xl font-extrabold tracking-tight text-[#111111]">
              makeitup
            </span>
            <span className="hidden sm:block text-[9px] tracking-[0.25em] uppercase text-[#777] -mt-1 font-sans">
              Make it yours. Make it up.
            </span>
          </div>

          {/* Right: Actions */}
          <div className="flex items-center gap-2 sm:gap-3">
            <button
              onClick={() => {
                setIsTrackOrderOpen(true);
                if (savedOrders.length > 0 && !trackedOrder) {
                  setTrackedOrder(savedOrders[0]);
                  setTrackingInputId(savedOrders[0].orderId);
                }
              }}
              className="p-2 rounded-full text-[#111111] hover:bg-black/5 md:hidden"
              aria-label="Track Order"
            >
              <Truck size={20} className="text-black" />
            </button>

            <button
              onClick={() => setIsSearchOpen(true)}
              className="p-2 rounded-full text-[#111111] hover:bg-black/5 transition-colors"
              aria-label="Search items"
            >
              <Search size={20} />
            </button>

            <button
              onClick={() => setIsWishlistOpen(true)}
              className="p-2 rounded-full text-[#111111] hover:bg-black/5 transition-colors relative"
              aria-label="Wishlist"
            >
              <Heart size={20} className={wishlist.length > 0 ? "fill-[#E0A800] text-[#E0A800]" : ""} />
              {wishlist.length > 0 && (
                <span className="absolute top-1 right-1 bg-[#111111] text-[#FAF7F0] text-[10px] w-4 h-4 rounded-full flex items-center justify-center font-bold">
                  {wishlist.length}
                </span>
              )}
            </button>

            <button
              onClick={() => setIsCartOpen(true)}
              className="p-2 sm:px-3 sm:py-2 bg-[#111111] text-[#FAF7F0] rounded-full hover:bg-black transition-colors flex items-center gap-2 relative shadow-md"
              aria-label="Cart"
            >
              <ShoppingBag size={18} className="text-[#E0A800]" />
              <span className="text-xs font-semibold hidden sm:inline">CART</span>
              <span className="bg-[#E0A800] text-[#111111] text-[11px] font-bold px-1.5 py-0.2 rounded-full">
                {totalCartItems}
              </span>
            </button>
          </div>

        </div>
      </header>

      {/* MOBILE SLIDE-OUT DRAWER */}
      {isMobileMenuOpen && (
        <div className="fixed inset-0 z-50 flex">
          <div className="fixed inset-0 bg-black/60 backdrop-blur-sm" onClick={() => setIsMobileMenuOpen(false)} />
          <div className="relative w-4/5 max-w-xs bg-[#FAF7F0] h-full shadow-2xl p-6 flex flex-col justify-between overflow-y-auto">
            <div>
              <div className="flex items-center justify-between pb-6 border-b border-[#E8E2D5]">
                <span className="font-serif text-2xl font-black">makeitup</span>
                <button onClick={() => setIsMobileMenuOpen(false)} className="p-1 rounded-full hover:bg-black/5">
                  <X size={22} />
                </button>
              </div>

              <div className="py-6 space-y-4">
                <p className="text-[10px] font-bold tracking-widest text-[#888] uppercase">Explore Categories</p>
                {CATEGORIES_NAV.map((cat) => (
                  <button
                    key={cat.key}
                    onClick={() => {
                      setSelectedCategory(cat.key as Category);
                      setIsMobileMenuOpen(false);
                      window.scrollTo({ top: 600, behavior: 'smooth' });
                    }}
                    className="flex items-center justify-between w-full text-left font-serif text-lg py-1.5 hover:text-[#E0A800] transition-colors"
                  >
                    <span>{cat.name}</span>
                    <span className="text-xs font-sans text-gray-500">{cat.count}</span>
                  </button>
                ))}
              </div>

              <div className="pt-4 border-t border-[#E8E2D5] space-y-3">
                <button
                  onClick={() => {
                    setIsTrackOrderOpen(true);
                    setIsMobileMenuOpen(false);
                  }}
                  className="w-full text-left text-xs uppercase tracking-wider text-black flex items-center gap-2 hover:text-[#E0A800] font-bold py-1"
                >
                  <Truck size={16} className="text-[#E0A800]" /> Track My Order
                </button>
                <a
                  href={`https://wa.me/${WHATSAPP_PHONE}?text=Hi%20makeitup%20team,%20I%20have%20an%20order%20inquiry!`}
                  target="_blank"
                  rel="noopener noreferrer"
                  className="w-full text-left text-xs uppercase tracking-wider text-green-700 flex items-center gap-2 font-semibold"
                >
                  <MessageCircle size={16} /> WhatsApp Concierge
                </a>
              </div>
            </div>

            <div className="text-xs text-gray-400 pt-6">
              © {new Date().getFullYear()} makeitup fashion. All rights reserved.
            </div>
          </div>
        </div>
      )}

      {/* SEARCH OVERLAY */}
      {isSearchOpen && (
        <div className="fixed inset-0 z-50 flex items-start justify-center pt-20 px-4">
          <div className="fixed inset-0 bg-black/70 backdrop-blur-sm" onClick={() => setIsSearchOpen(false)} />
          <div className="relative w-full max-w-2xl bg-[#FAF7F0] rounded-2xl shadow-2xl p-6 border border-[#E8E2D5] z-10">
            <div className="flex items-center justify-between pb-4 border-b border-[#E8E2D5]">
              <div className="flex items-center gap-3 w-full">
                <Search size={22} className="text-[#E0A800]" />
                <input
                  type="text"
                  placeholder="Search shirts, kurta, trousers, sunglasses..."
                  value={searchQuery}
                  onChange={(e) => setSearchQuery(e.target.value)}
                  autoFocus
                  className="w-full bg-transparent text-lg focus:outline-none placeholder-gray-400 font-serif"
                />
              </div>
              <button onClick={() => setIsSearchOpen(false)} className="p-2 text-gray-500 hover:text-black">
                <X size={20} />
              </button>
            </div>

            <div className="pt-4 max-h-96 overflow-y-auto space-y-3">
              {searchQuery.trim() ? (
                filteredProducts.length > 0 ? (
                  filteredProducts.map((product) => (
                    <div
                      key={product.id}
                      onClick={() => {
                        setSelectedProduct(product);
                        setIsSearchOpen(false);
                      }}
                      className="flex items-center gap-4 p-2 rounded-xl hover:bg-black/5 cursor-pointer transition-colors"
                    >
                      <img src={product.imageUrl} alt={product.name} className="w-14 h-14 object-cover rounded-lg" />
                      <div className="flex-1">
                        <p className="text-xs uppercase text-[#E0A800] font-bold tracking-wider">{product.category}</p>
                        <h4 className="font-serif font-bold text-sm">{product.name}</h4>
                        <p className="text-xs font-semibold text-gray-900">₹{product.price.toLocaleString('en-IN')}</p>
                      </div>
                      <ArrowRight size={18} className="text-gray-400" />

                      src={item.product.imageUrl}
                      alt={item.product.name}
                      className="w-20 h-24 object-cover rounded-xl bg-gray-100"
                    />
                    <div className="flex-1 flex flex-col justify-between">
                      <div>
                        <div className="flex items-start justify-between">
                          <h4 className="font-serif font-bold text-sm text-gray-900 line-clamp-1">
                            {item.product.name}
                          </h4>
                          <button
                            onClick={() => removeCartItem(item.product.id, item.selectedSize)}
                            className="text-gray-400 hover:text-red-500 p-0.5"
                          >
                            <X size={16} />
                          </button>
                        </div>
                        <span className="text-[10px] font-semibold bg-gray-100 text-gray-700 px-2 py-0.5 rounded">
                          Size: {item.selectedSize}
                        </span>
                      </div>

                      <div className="flex items-center justify-between mt-2">
                        <div className="flex items-center border border-gray-200 rounded-lg overflow-hidden bg-gray-50">
                          <button
                            onClick={() => updateCartQuantity(item.product.id, item.selectedSize, -1)}
                            className="px-2 py-1 text-xs hover:bg-gray-200 text-gray-700 font-bold"
                          >
                            -
                          </button>
                          <span className="px-2.5 text-xs font-semibold">{item.quantity}</span>
                          <button
                            onClick={() => updateCartQuantity(item.product.id, item.selectedSize, 1)}
                            className="px-2 py-1 text-xs hover:bg-gray-200 text-gray-700 font-bold"
                          >
                            +
                          </button>
                        </div>

                        <span className="text-sm font-bold text-black">
                          ₹{(item.product.price * item.quantity).toLocaleString('en-IN')}
                        </span>
                      </div>
                    </div>
                  </div>
                ))
              )}
            </div>

            {cart.length > 0 && (
              <div className="p-6 border-t border-[#E8E2D5] bg-white space-y-4">
                <div className="space-y-1">
                  <div className="flex gap-2">
                    <input
                      type="text"
                      placeholder="Coupon: MAKEITUP10"
                      value={couponCode}
                      onChange={(e) => setCouponCode(e.target.value)}
                      className="flex-1 px-3 py-2 text-xs border border-gray-300 rounded-xl uppercase font-medium focus:outline-none focus:border-black"
                    />
                    <button
                      onClick={applyCoupon}
                      className="px-4 py-2 bg-black text-white text-xs font-bold rounded-xl hover:bg-[#E0A800] hover:text-black transition-colors"
                    >
                      Apply
                    </button>
                  </div>
                  {couponError && <p className="text-[11px] text-red-500 font-medium">{couponError}</p>}
                  {appliedCoupon && (
                    <p className="text-[11px] text-green-700 font-semibold">
                      ✓ Coupon {appliedCoupon.code} applied ({appliedCoupon.discountPercent}% off)
                    </p>
                  )}
                </div>

                <div className="space-y-1.5 text-xs pt-2">
                  <div className="flex justify-between text-gray-600">
                    <span>Subtotal</span>
                    <span>₹{cartSubtotal.toLocaleString('en-IN')}</span>
                  </div>
                  {appliedCoupon && (
                    <div className="flex justify-between text-green-700 font-semibold">
                      <span>Discount ({appliedCoupon.discountPercent}%)</span>
                      <span>-₹{discountAmount.toLocaleString('en-IN')}</span>
                    </div>
                  )}
                  <div className="flex justify-between text-gray-600">
                    <span>Delivery</span>
                    <span>{shippingFee === 0 ? <strong className="text-green-700">FREE</strong> : `₹${shippingFee}`}</span>
                  </div>
                  <div className="flex justify-between text-base font-extrabold text-black pt-2 border-t border-gray-100">
                    <span>Grand Total</span>
                    <span>₹{cartTotal.toLocaleString('en-IN')}</span>
                  </div>
                </div>

                <button
                  onClick={() => {
                    setIsCartOpen(false);
                    setIsCheckoutOpen(true);
                  }}
                  className="w-full py-4 bg-[#111111] text-[#FAF7F0] hover:bg-[#E0A800] hover:text-black font-bold text-xs uppercase tracking-widest rounded-xl transition-all shadow-xl flex items-center justify-center gap-2"
                >
                  <span>Proceed to Checkout</span>
                  <ArrowRight size={16} />
                </button>
              </div>
            )}

          </div>
        </div>
      )}

      {/* ========================================================================= */}
      {/* 4. WISHLIST DRAWER */}
      {/* ========================================================================= */}
      {isWishlistOpen && (
        <div className="fixed inset-0 z-50 flex justify-end">
          <div className="fixed inset-0 bg-black/60 backdrop-blur-sm" onClick={() => setIsWishlistOpen(false)} />
          <div className="relative w-full max-w-md bg-[#FAF7F0] h-full shadow-2xl flex flex-col justify-between z-10">
            
            <div className="p-6 border-b border-[#E8E2D5] flex items-center justify-between">
              <div className="flex items-center gap-2">
                <Heart size={20} className="fill-[#E0A800] text-[#E0A800]" />
                <h3 className="font-serif text-xl font-bold">Saved Items</h3>
                <span className="text-xs bg-[#111111] text-[#FAF7F0] px-2 py-0.5 rounded-full font-bold">
                  {wishlist.length}
                </span>
              </div>
              <button onClick={() => setIsWishlistOpen(false)} className="p-1 rounded-full hover:bg-black/5">
                <X size={20} />
              </button>
            </div>

            <div className="flex-1 overflow-y-auto p-6 space-y-4">
              {wishlist.length === 0 ? (
                <div className="text-center py-16">
                  <Heart size={44} className="mx-auto text-gray-300 mb-3" />
                  <p className="font-serif text-lg font-bold text-gray-800">Your wishlist is empty</p>
                  <p className="text-xs text-gray-500 mt-1 mb-6">Tap the heart on any piece to save it for later.</p>
                </div>
              ) : (
                PRODUCTS.filter((p) => wishlist.includes(p.id)).map((product) => (
                  <div
                    key={product.id}
                    className="flex gap-4 p-3 bg-white rounded-2xl border border-[#E8E2D5] shadow-xs"
                  >
                    <img
                      src={product.imageUrl}
                      alt={product.name}
                      className="w-18 h-22 object-cover rounded-xl"
                    />
                    <div className="flex-1 flex flex-col justify-between">
                      <div>
                        <div className="flex items-start justify-between">
                          <h4 className="font-serif font-bold text-sm text-gray-900 line-clamp-1">{product.name}</h4>
                          <button onClick={() => toggleWishlist(product.id)} className="text-gray-400 hover:text-red-500">
                            <X size={16} />
                          </button>
                        </div>
                        <p className="text-xs font-bold text-black mt-1">₹{product.price.toLocaleString('en-IN')}</p>
                      </div>

                      <button
                        onClick={() => {
                          addToCart(product);
                          toggleWishlist(product.id);
                        }}
                        className="mt-2 py-1.5 bg-[#111111] text-[#FAF7F0] text-xs font-bold rounded-lg hover:bg-[#E0A800] hover:text-black transition-colors"
                      >
                        Move to Bag
                      </button>
                    </div>
                  </div>
                ))
              )}
            </div>

            <div className="p-6 border-t border-[#E8E2D5] bg-white">
              <button
                onClick={() => setIsWishlistOpen(false)}
                className="w-full py-3 border border-black rounded-xl text-xs font-bold uppercase tracking-wider hover:bg-black hover:text-white transition-colors"
              >
                Continue Browsing
              </button>
            </div>

          </div>
        </div>
      )}

      {/* ========================================================================= */}
      {/* 5. CHECKOUT MODAL */}
      {/* ========================================================================= */}
      {isCheckoutOpen && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
          <div className="fixed inset-0 bg-black/70 backdrop-blur-sm" onClick={() => setIsCheckoutOpen(false)} />
          <div className="relative w-full max-w-2xl bg-[#FAF7F0] rounded-3xl shadow-2xl overflow-hidden border border-[#E8E2D5] z-10 max-h-[95vh] flex flex-col">
            
            <div className="p-6 border-b border-[#E8E2D5] flex items-center justify-between bg-white">
              <div>
                <span className="text-[10px] font-bold uppercase tracking-widest text-[#E0A800]">Fast Checkout</span>
                <h3 className="font-serif text-2xl font-bold">Delivery & Payment</h3>
              </div>
              <button onClick={() => setIsCheckoutOpen(false)} className="p-1 rounded-full hover:bg-black/5">
                <X size={20} />
              </button>
            </div>

            <form onSubmit={handlePlaceOrder} className="flex-1 overflow-y-auto p-6 sm:p-8 space-y-6">
              
              <div className="bg-[#F4EFE6] p-4 rounded-2xl border border-[#E8E2D5] flex items-center justify-between text-xs">
                <div>
                  <span className="font-semibold text-gray-800">{totalCartItems} item(s) in bag</span>
                  <p className="text-gray-500">Shipping to your verified address</p>
                </div>
                <div className="text-right">
                  <span className="text-base font-black text-black">₹{cartTotal.toLocaleString('en-IN')}</span>
                  {shippingFee === 0 && <p className="text-[10px] text-green-700 font-bold">FREE Delivery</p>}
                </div>
              </div>

              <div className="space-y-4">
                <p className="text-xs font-bold uppercase tracking-widest text-black">1. Customer Details</p>
                
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs font-semibold text-gray-700 mb-1">Full Name *</label>
                    <input
                      type="text"
                      placeholder="e.g. Rahul Sharma"
                      value={checkoutForm.name}
                      onChange={(e) => setCheckoutForm({ ...checkoutForm, name: e.target.value })}
                      className="w-full px-3.5 py-2.5 text-xs bg-white border border-gray-300 rounded-xl focus:outline-none focus:border-black"
                    />
                    {formErrors.name && <p className="text-[10px] text-red-500 mt-1">{formErrors.name}</p>}
                  </div>

                  <div>
                    <label className="block text-xs font-semibold text-gray-700 mb-1">Mobile Phone (10 digits) *</label>
                    <input
                      type="tel"
                      placeholder="e.g. 9876543210"
                      value={checkoutForm.phone}
                      onChange={(e) => setCheckoutForm({ ...checkoutForm, phone: e.target.value })}
                      className="w-full px-3.5 py-2.5 text-xs bg-white border border-gray-300 rounded-xl focus:outline-none focus:border-black"
                    />
                    {formErrors.phone && <p className="text-[10px] text-red-500 mt-1">{formErrors.phone}</p>}
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-gray-700 mb-1">Email Address (Order Confirmation) *</label>
                  <input
                    type="email"
                    placeholder="e.g. rahul@gmail.com"
                    value={checkoutForm.email}
                    onChange={(e) => setCheckoutForm({ ...checkoutForm, email: e.target.value })}
                    className="w-full px-3.5 py-2.5 text-xs bg-white border border-gray-300 rounded-xl focus:outline-none focus:border-black"
                  />
                  {formErrors.email && <p className="text-[10px] text-red-500 mt-1">{formErrors.email}</p>}
                </div>
              </div>

              <div className="space-y-4 pt-2">
                <p className="text-xs font-bold uppercase tracking-widest text-black">2. Shipping Address</p>

                <div>
                  <label className="block text-xs font-semibold text-gray-700 mb-1">Flat / House No. / Street Address *</label>
                  <input
                    type="text"
                    placeholder="e.g. Flat 402, Signature Heights, MG Road"
                    value={checkoutForm.address}
                    onChange={(e) => setCheckoutForm({ ...checkoutForm, address: e.target.value })}
                    className="w-full px-3.5 py-2.5 text-xs bg-white border border-gray-300 rounded-xl focus:outline-none focus:border-black"
                  />
                  {formErrors.address && <p className="text-[10px] text-red-500 mt-1">{formErrors.address}</p>}
                </div>

                <div className="grid grid-cols-2 gap-4">
                  <div>
                    <label className="block text-xs font-semibold text-gray-700 mb-1">City *</label>
                    <input
                      type="text"
                      placeholder="e.g. Mumbai"
                      value={checkoutForm.city}
                      onChange={(e) => setCheckoutForm({ ...checkoutForm, city: e.target.value })}
                      className="w-full px-3.5 py-2.5 text-xs bg-white border border-gray-300 rounded-xl focus:outline-none focus:border-black"
                    />
                    {formErrors.city && <p className="text-[10px] text-red-500 mt-1">{formErrors.city}</p>}
                  </div>

                  <div>
                    <label className="block text-xs font-semibold text-gray-700 mb-1">Postal Pincode (6 digits) *</label>
                    <input
                      type="text"
                      maxLength={6}
                      placeholder="e.g. 400001"
                      value={checkoutForm.pincode}
                      onChange={(e) => setCheckoutForm({ ...checkoutForm, pincode: e.target.value })}
                      className="w-full px-3.5 py-2.5 text-xs bg-white border border-gray-300 rounded-xl focus:outline-none focus:border-black"
                    />
                    {formErrors.pincode && <p className="text-[10px] text-red-500 mt-1">{formErrors.pincode}</p>}
                  </div>
                </div>

                <div>
                  <label className="block text-xs font-semibold text-gray-700 mb-1">Order / Landmark Notes (Optional)</label>
                  <input
                    type="text"
                    placeholder="e.g. Leave with security / Ring bell twice"
                    value={checkoutForm.notes}
                    onChange={(e) => setCheckoutForm({ ...checkoutForm, notes: e.target.value })}
                    className="w-full px-3.5 py-2.5 text-xs bg-white border border-gray-300 rounded-xl focus:outline-none focus:border-black"
                  />
                </div>
              </div>

              <div className="space-y-3 pt-2">
                <p className="text-xs font-bold uppercase tracking-widest text-black">3. Payment Mode</p>
                <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
                  <label
                    className={`flex items-start gap-3 p-3.5 rounded-2xl border cursor-pointer transition-all ${
                      checkoutForm.paymentMethod === 'COD'
                        ? 'border-black bg-white shadow-sm ring-2 ring-black'
                        : 'border-gray-200 bg-white/60 hover:border-gray-400'
                    }`}
                  >
                    <input
                      type="radio"
                      name="paymentMethod"
                      checked={checkoutForm.paymentMethod === 'COD'}
                      onChange={() => setCheckoutForm({ ...checkoutForm, paymentMethod: 'COD' })}
                      className="mt-0.5 accent-black"
                    />
                    <div>
                      <p className="text-xs font-bold text-black">Cash on Delivery (COD)</p>
                      <p className="text-[11px] text-gray-500 mt-0.5">Pay via Cash or Card on door arrival</p>
                    </div>
                  </label>

                  <label
                    className={`flex items-start gap-3 p-3.5 rounded-2xl border cursor-pointer transition-all ${
                      checkoutForm.paymentMethod === 'UPI'
                        ? 'border-black bg-white shadow-sm ring-2 ring-black'
                        : 'border-gray-200 bg-white/60 hover:border-gray-400'
                    }`}
                  >
                    <input
                      type="radio"
                      name="paymentMethod"
                      checked={checkoutForm.paymentMethod === 'UPI'}
                      onChange={() => setCheckoutForm({ ...checkoutForm, paymentMethod: 'UPI' })}
                      className="mt-0.5 accent-black"
                    />
                    <div>
                      <p className="text-xs font-bold text-black">Instant UPI / QR</p>
                      <p className="text-[11px] text-gray-500 mt-0.5">GPay, PhonePe, Paytm QR code on submit</p>
                    </div>
                  </label>
                </div>

                {checkoutForm.paymentMethod === 'UPI' && (
                  <div className="bg-amber-50 border border-amber-200 p-3.5 rounded-xl text-xs text-amber-900">
                    <p className="font-semibold">UPI Payment Instructions:</p>
                    <p className="mt-0.5 text-[11px]">
                      Send <strong>₹{cartTotal.toLocaleString('en-IN')}</strong> to UPI ID: <code className="bg-amber-100 px-1.5 py-0.5 rounded font-mono font-bold">makeitup@upi</code> or scan the order confirmation screen QR.
                    </p>
                  </div>
                )}
              </div>

              <div className="pt-4 border-t border-gray-200">
                <button
                  type="submit"
                  disabled={isSubmittingOrder}
                  className="w-full py-4 bg-[#111111] text-[#FAF7F0] hover:bg-[#E0A800] hover:text-black font-bold text-xs uppercase tracking-widest rounded-xl transition-all shadow-xl flex items-center justify-center gap-2 disabled:opacity-50"
                >
                  {isSubmittingOrder ? (
                    <span className="flex items-center gap-2">
                      <Sparkles className="animate-spin" size={16} /> Transmitting Order...
                    </span>
                  ) : (
                    <>
                      <span>Place Order (₹{cartTotal.toLocaleString('en-IN')})</span>
                      <ArrowRight size={16} />
                    </>
                  )}
                </button>
                <p className="text-[11px] text-gray-500 text-center mt-2">
                  Order details emailed automatically to <strong>{ORDER_TARGET_EMAIL}</strong> and saved to your device.
                </p>
              </div>

            </form>

          </div>
        </div>
      )}

      {/* ========================================================================= */}
      {/* 6. ORDER CONFIRMATION SCREEN */}
      {/* ========================================================================= */}
      {placedOrder && (
        <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
          <div className="fixed inset-0 bg-black/75 backdrop-blur-sm" onClick={() => setPlacedOrder(null)} />
          <div className="relative w-full max-w-lg bg-[#FAF7F0] rounded-3xl shadow-2xl overflow-hidden border border-[#E8E2D5] z-10 max-h-[95vh] flex flex-col p-6 sm:p-8">
            
            <div className="text-center pb-6 border-b border-[#E8E2D5]">
              <div className="w-14 h-14 bg-green-100 text-green-700 rounded-full flex items-center justify-center mx-auto mb-3">
                <CheckCircle2 size={32} />
              </div>
              <span className="text-[10px] font-bold uppercase tracking-widest text-[#E0A800]">Order Confirmed</span>
              <h3 className="font-serif text-2xl font-black mt-1">Thank You, {placedOrder.customer.name}!</h3>
              <p className="text-xs text-gray-600 mt-1">
                Order ID: <strong className="text-black font-mono">{placedOrder.orderId}</strong>
              </p>
            </div>

            <div className="flex-1 overflow-y-auto py-6 space-y-4 text-xs">
              <div className="bg-white p-4 rounded-2xl border border-[#E8E2D5] space-y-2">
                <div className="flex justify-between font-bold text-gray-900 border-b border-gray-100 pb-2">
                  <span>Purchased Pieces</span>
                  <span>Amount</span>
                </div>
                {placedOrder.items.map((i, idx) => (
                  <div key={idx} className="flex justify-between text-gray-600">
                    <span>{i.product.name} ({i.selectedSize}) x {i.quantity}</span>
                    <span className="font-semibold text-black">₹{(i.product.price * i.quantity).toLocaleString('en-IN')}</span>
                  </div>
                ))}
                <div className="border-t border-gray-100 pt-2 flex justify-between font-bold text-black text-sm">
                  <span>Total Paid</span>
                  <span>₹{placedOrder.total.toLocaleString('en-IN')} ({placedOrder.customer.paymentMethod})</span>
                </div>
              </div>

              <div className="bg-white p-4 rounded-2xl border border-[#E8E2D5]">
                <p className="font-bold text-black mb-1">Delivery Address</p>
                <p className="text-gray-600 leading-relaxed">
                  {placedOrder.customer.address}, {placedOrder.customer.city} - {placedOrder.customer.pincode}
                </p>
                <p className="text-gray-500 mt-1">Phone: {placedOrder.customer.phone}</p>
              </div>

              <div className="bg-amber-50 p-4 rounded-2xl border border-amber-200 text-amber-900">
                <p className="font-bold">Dispatch Notification Sent</p>
                <p className="mt-1 text-[11px] leading-relaxed">
                  Full details have been automatically emailed to <strong>{ORDER_TARGET_EMAIL}</strong> and customer email <strong>{placedOrder.customer.email}</strong>. Estimated delivery: {placedOrder.estimatedDelivery}.
                </p>
              </div>
            </div>

            <div className="pt-4 border-t border-[#E8E2D5] flex gap-3">
              <button
                onClick={() => {
                  setTrackedOrder(placedOrder);
                  setTrackingInputId(placedOrder.orderId);
                  setPlacedOrder(null);
                  setIsTrackOrderOpen(true);
                }}
                className="flex-1 py-3 bg-[#E0A800] text-black rounded-xl font-bold text-xs uppercase tracking-wider text-center flex items-center justify-center gap-2 hover:bg-black hover:text-[#FAF7F0] transition-colors"
              >
                <Truck size={16} /> Track Order Now
              </button>
              <button
                onClick={() => setPlacedOrder(null)}
                className="px-6 py-3 bg-[#111111] text-[#FAF7F0] rounded-xl font-bold text-xs uppercase tracking-wider hover:bg-black transition-colors"
              >
                Done
              </button>
            </div>

          </div>
        </div>
      )}

    </div>
  );
}
